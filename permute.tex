% every ordered selection of #2 pairwise distinct entries, in the order of
% python's itertools.permutations - so no entry twice, no selection twice, and
% nothing at all if #2 is longer than the list.
% (heap's algorithm plus truncation, which i had here before, gives none of that)
%
% #1 the name of the list, please define '\chr@@output' to run on the list after
% each selection. this macro receives no arguments, but can read the same list
% you gave to permute (its first #2 entries hold the current selection).
% besides, it can use '\chr@list@coll' to get the collected list elements!
% #2 is the length of the selections desired
\def\chr@permute#1#2{\begingroup
   \ifx\chr@@output\chr@undefined
      \chr@error{You have to define \string\chr@@output\space to run on the list after each permutation!}%
   \fi
   \def\chr@permute@listname{#1}%
   \edef\chr@permute@output@length{\number#2}%
   \edef\chr@permute@input@length{\number\csname#1length\endcsname}%
   \let\chr@list@coll=\chr@empty
   \chr@verbose{Enumerate all \chr@permute@output@length-selections of list #1
      (length: \chr@permute@input@length).}%
   \ifnum\chr@permute@output@length>\chr@permute@input@length\relax
      \chr@verbose{    None, the head is longer than the store.}%
   \else
      \chr@permute@mksucc
      \chr@permute@snapshot{#1}%
      \chr@permute@select{0}%
   \fi
\endgroup}

% without numexpr this is like... really painful, so: a successor table
\def\chr@permute@mksucc{%
   \chr@arith=\z@
   \loop
      \chr@tempcountb=\chr@arith \chr@step\chr@tempcountb
      \expandafter\edef\csname chr@succ@\the\chr@arith\endcsname{\the\chr@tempcountb}%
      \chr@step\chr@arith
      \ifnum\chr@arith<\chr@permute@output@length
   \repeat
}

% the list is rewritten in place, so keep the original order around
\def\chr@permute@snapshot#1{%
   \ifnum\chr@permute@input@length>\z@\relax
      \chr@arith=\z@
      \loop
         \chr@list@let\chr@perm@tmp=#1[\the\chr@arith]%
         \expandafter\let\csname chr@perm@src@\the\chr@arith\endcsname=\chr@perm@tmp
         \chr@step\chr@arith
         \ifnum\chr@arith<\chr@permute@input@length
      \repeat
   \fi
}

% #1 = current depth, pick the element for position #1
\def\chr@permute@select#1{%
   \ifnum#1<\chr@permute@output@length\relax
      \expandafter\def\csname chr@perm@j@#1\endcsname{0}%
      \chr@permute@loop{#1}%
   \else
      \chr@permute@emit
   \fi
}

% walk the candidate source indices for depth #1
\def\chr@permute@loop#1{%
   \edef\chr@perm@jcur{\csname chr@perm@j@#1\endcsname}%
   \ifnum\chr@perm@jcur<\chr@permute@input@length\relax
      \expandafter\chr@permute@loop@body
   \else
      \expandafter\chr@gobble
   \fi
   {#1}%
}
\def\chr@permute@loop@body#1{%
   \expandafter\ifx\csname chr@perm@used@\chr@perm@jcur\endcsname\relax
      \chr@permute@descend{#1}%
   \fi
   % the recursion clobbered \chr@perm@jcur
   \edef\chr@perm@jcur{\csname chr@perm@j@#1\endcsname}%
   \chr@incr\chr@perm@jcur
   \expandafter\edef\csname chr@perm@j@#1\endcsname{\chr@perm@jcur}%
   \chr@permute@loop{#1}%
}
\def\chr@permute@descend#1{%
   \expandafter\let\csname chr@perm@used@\chr@perm@jcur\endcsname=\chr@ns
   \expandafter\let\expandafter\chr@perm@tmp\expandafter
      =\csname chr@perm@src@\chr@perm@jcur\endcsname
   \expandafter\let\csname\chr@permute@listname @#1\endcsname=\chr@perm@tmp
   \edef\chr@perm@nd{\csname chr@succ@#1\endcsname}%
   \expandafter\chr@permute@select\expandafter{\chr@perm@nd}%
   \edef\chr@perm@jcur{\csname chr@perm@j@#1\endcsname}%
   \expandafter\let\csname chr@perm@used@\chr@perm@jcur\endcsname=\relax
}

\def\chr@permute@emit{%
   \expandafter\chr@list@to@string\expandafter\chr@list@coll
      \expandafter{\chr@permute@listname}{\chr@permute@output@length}%
   \chr@@output
}
