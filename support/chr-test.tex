% shared helpers for the l3build checks
\makeatletter
% write the current constraint store to the log
\def\SHOWSTORE{%
   \chr@list@to@string\chr@test@store{chr@constraints}%
      {\csname chr@constraintslength\endcsname}%
   \TYPE{STORE: [\chr@test@store]}%
}
% n <= m  <=>  n < m+1, tex has no \ifnumleq
\def\leq#1#2{\chr@tempcountb=#2\relax\advance\chr@tempcountb by\@ne\ifnum#1<\chr@tempcountb}
\def\sub#1#2{\chr@tempcount=#2\relax\advance\chr@tempcount by -#1\relax\edef\res{\the\chr@tempcount}}
\def\tuplehelper#1:#2\@nil{\def\fst{#1}\def\snd{#2}}
\def\tuple#1{\edef\@tmp{#1}\expandafter\tuplehelper\@tmp\@nil}
\def\add#1#2{\chr@tempcount=#1\relax\advance\chr@tempcount by #2\relax\edef\res{\the\chr@tempcount}}
\makeatother
